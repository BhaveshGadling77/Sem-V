  int spinlock; // to protect the queue
}
void block(lock *l) {
    myproc()->status = WAITING;
    spinlock(l->spinlock);
    move proc to l->q;  // there is a race here!
    spinunlock(l->spinlock);
    sched(); 
}
void lock(lock *l) {
 while(testandset(l->locked))
    block(l);

void unlock(lock *l) {
  spinlock(l->spinlock);
  remove all processes from l->q; // imp
  spinunlock(l->spinlock);
  loop: proc->status = RUNNABLE;
  l->locked = 0;
}
